{%
     \ifcase#1 % select font size
          \or\selffamily AA
          \or\selffamily AB
          \or\selffamily AC
          \or\selffamily AD
          \or\selffamily AE
          \or\selffamily AF
          \or\selffamily AG
          \or\selffamily BA
          \else\selffamily AC
     \fi
     \ifcase#2 % select font face
          \or\rm
          \or\it
          \or\bf
          \or\bi
          \else\rm
     \fi
}
