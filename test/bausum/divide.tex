\def\fixdimens#1#2% Want #1*#2/1000 and (#1*#2/1000)*1.2
{%
  \bgroup
     \dimen1=#1pt
     \count2=#2
     \count1=\dimen1
     \multiply\count1 by \count2
     \divide\count1 by 1000
     \dimen1=\count1 sp
     \dimen2=1.2\dimen1% Note, `\multiply\dimen1 by 1.2' generates an error!
     Original point size = #1pt with  multiplier = #2 makes:
     \the\dimen1/\the\dimen2.\par
  \egroup
}
\count1=1
Before fixdimens, count1 = \the\count1.\par
\fixdimens{10}{1000}
\fixdimens{10}{968}
\fixdimens{24}{1044}
After fixdimens, count1 = \the\count1.
\bye
