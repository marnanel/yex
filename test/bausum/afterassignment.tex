\def\ttrD
{%
     \advance\tnotenumber by 1%
     \unskip
     \if\next.% the working version has other special cases: e.g., `)' `,'.
          .\superscript{\the\tnotenumber}%
     \else
          \R1\superscript{\the\tnotenumber}\ \next
     \fi
     \global\setbox\tnotebox=\hbox
     {%
          \noindent
          \unhbox\tnotebox
          \superscript{\the\tnotenumber}%
          \R1\tpage.\quad
     }%
}
