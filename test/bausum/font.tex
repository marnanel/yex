\def\tmkfont#1#2#3% filename; point size; scale factor;
{%
     \def\emptyB{#2}%
     \def\emptyC{#3}%
     \ifx\emptyB\empty
          \ifx\emptyC\empty
               \global\font\tff=#1 % The space in `#1 %' is required!
          \else
               \global\font\tff=#1 scaled #3 %
          \fi
     \else
          \global\font\tff=#1 at #2 %
     \fi
     . \tff\fontname\tff&
     \tff The quick red fox jumped over the log.\cr
}
